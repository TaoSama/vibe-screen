from pathlib import Path
root = Path('.')
recorded = {}
for line in (root / 'SHA256SUMS').read_text().splitlines():
    digest, relative = line.split('  ./', maxsplit=1)
    recorded[relative] = digest
expected = sorted(path.relative_to(root).as_posix() for path in root.rglob('*') if path.is_file() and path.name != 'SHA256SUMS')
missing = sorted(set(expected) - set(recorded))
extra = sorted(set(recorded) - set(expected))
bad = [relative for relative, digest in recorded.items() if relative in expected and sha256((root / relative).read_bytes()).hexdigest() != digest]
print({'expected_count': len(expected), 'recorded_count': len(recorded), 'missing': missing, 'extra': extra, 'bad': bad})
raise SystemExit(1 if missing or extra or bad else 0)
